    <canvas id="gameCanvas" width="800" height="300"></canvas>
    
    <!-- Level Up Selection UI -->
    <div id="cardOverlay">
      <h2 style="margin:0; color:#ffd700;">LEVEL UP! CHOOSE AN UPGRADE</h2>
      <div class="cards-container" id="cardsContainer"></div>
    </div>
  </div>

  <div class="instructions">SPACE / TAP to Jump. Double Jump unlocked via upgrades!</div>

<script>
const canvas = document.getElementById('gameCanvas');
const ctx = canvas.getContext('2d');
const cardOverlay = document.getElementById('cardOverlay');
const cardsContainer = document.getElementById('cardsContainer');

// Game State
let lastTime = performance.now();
let distance = 0;
let level = 1;
let currentXP = 0;
let targetXP = 100;
let gameOver = false;
let pausedForUpgrade = false;
let gameSpeed = 320;

// Player Attributes (Upgradeable)
const dino = {
  x: 50,
  y: 200,
  width: 40,
  height: 50,
  velocityY: 0,
  gravity: 1600,
  jumpForce: -580,
  grounded: true,
  maxJumps: 1,
  jumpsRemaining: 1,
  shieldCount: 0,
  magnetRadius: 0,
  xpMultiplier: 1.0,
  scoreMultiplier: 1.0
};

// Item Pool Database
const UPGRADE_POOL = [
  // Common
  { id: 'jump_boost', name: 'Spring Boots', desc: '+15% Jump Height', rarity: 'common', apply: () => dino.jumpForce -= 80 },
  { id: 'xp_boost', name: 'Fast Learner', desc: '+30% XP Gain', rarity: 'common', apply: () => dino.xpMultiplier += 0.3 },
  { id: 'score_boost', name: 'Greed', desc: '+25% Score Multiplier', rarity: 'common', apply: () => dino.scoreMultiplier += 0.25 },
  // Rare
  { id: 'double_jump', name: 'Wings', desc: 'Grants +1 Extra Air Jump', rarity: 'rare', apply: () => dino.maxJumps++ },
  { id: 'shield', name: 'Energy Barrier', desc: 'Absorbs 1 Collision Hit', rarity: 'rare', apply: () => dino.shieldCount++ },
  { id: 'magnet', name: 'Attractor Field', desc: 'Pulls nearby XP orbs to you', rarity: 'rare', apply: () => dino.magnetRadius += 120 },
  // Legendary
  { id: 'time_slow', name: 'Chrono Warp', desc: 'Permanently reduces obstacle speed by 20%', rarity: 'legendary', apply: () => gameSpeed *= 0.8 },
  { id: 'aegis', name: 'Aegis Shield', desc: 'Gain +2 Hits Shield & +50% XP', rarity: 'legendary', apply: () => { dino.shieldCount += 2; dino.xpMultiplier += 0.5; } }
];

const obstacles = [];
const xpOrbs = [];

function spawnObstacle() {
  const height = 30 + Math.random() * 35;
  obstacles.push({
    x: canvas.width + 50,
    y: 250 - height,
    width: 22,
    height: height,
    active: true
  });
}

function spawnXPOrb() {
  xpOrbs.push({
    x: canvas.width + 50,
    y: 100 + Math.random() * 100,
    size: 8,
    value: 25,
    active: true
  });
}

let obstacleTimer = 0;
let orbTimer = 0;

// Input Handling
function triggerJump() {
  if (pausedForUpgrade) return;
  if (gameOver) { restartGame(); return; }

  if (dino.grounded || dino.jumpsRemaining > 0) {
    dino.velocityY = dino.jumpForce;
    dino.grounded = false;
    dino.jumpsRemaining--;
  }
}

window.addEventListener('keydown', (e) => {
  if (e.code === 'Space' || e.code === 'ArrowUp') {
    e.preventDefault();
    triggerJump();
  }
});
canvas.addEventListener('touchstart', (e) => {
  e.preventDefault();
  triggerJump();
}, { passive: false });

function restartGame() {
  distance = 0;
  level = 1;
  currentXP = 0;
  targetXP = 100;
  gameSpeed = 320;
  obstacles.length = 0;
  xpOrbs.length = 0;
  
  // Reset Stats
  dino.y = 200;
  dino.velocityY = 0;
  dino.jumpForce = -580;
  dino.maxJumps = 1;
  dino.shieldCount = 0;
  dino.magnetRadius = 0;
  dino.xpMultiplier = 1.0;
  dino.scoreMultiplier = 1.0;
  
  gameOver = false;
  pausedForUpgrade = false;
  cardOverlay.style.display = 'none';
  lastTime = performance.now();
  requestAnimationFrame(gameLoop);
}

// Roguelite Upgrade System
function addXP(amount) {
  currentXP += amount * dino.xpMultiplier;
  if (currentXP >= targetXP) {
    currentXP -= targetXP;
    level++;
    targetXP = Math.floor(targetXP * 1.5);
    triggerLevelUp();
  }
}

function triggerLevelUp() {
  pausedForUpgrade = true;
  cardsContainer.innerHTML = '';

  // Select 3 Unique Weighted Upgrades
  const choices = [];
  const poolCopy = [...UPGRADE_POOL];

  while (choices.length < 3 && poolCopy.length > 0) {
    const roll = Math.random();
    let targetRarity = 'common';
    if (roll > 0.95) targetRarity = 'legendary';
    else if (roll > 0.70) targetRarity = 'rare';

    let eligible = poolCopy.filter(u => u.rarity === targetRarity);
    if (eligible.length === 0) eligible = poolCopy; // Fallback

    const index = Math.floor(Math.random() * eligible.length);
    const selected = eligible[index];

    choices.push(selected);
    // Remove selected item from temporary roll pool
    poolCopy.splice(poolCopy.indexOf(selected), 1);
  }

  // Render UI Cards
  choices.forEach(upgrade => {
    const card = document.createElement('div');
    card.className = `card ${upgrade.rarity}`;
    card.innerHTML = `
      <div>
        <div class="rarity">${upgrade.rarity}</div>
        <h3>${upgrade.name}</h3>
      </div>
      <p>${upgrade.desc}</p>
    `;
    card.onclick = () => selectUpgrade(upgrade);
    cardsContainer.appendChild(card);
  });

  cardOverlay.style.display = 'flex';
}

function selectUpgrade(upgrade) {
  upgrade.apply();
  cardOverlay.style.display = 'none';
  pausedForUpgrade = false;
  lastTime = performance.now();
  requestAnimationFrame(gameLoop);
}

// Main Game Loop
function gameLoop(now) {
  if (gameOver || pausedForUpgrade) return;

  const dt = Math.min((now - lastTime) / 1000, 0.1);
  lastTime = now;

  update(dt);
  render();

  requestAnimationFrame(gameLoop);
}

function update(dt) {
  distance += gameSpeed * dt * 0.1 * dino.scoreMultiplier;
  gameSpeed += dt * 3; // Progressive difficulty scaling

  // Physics
  dino.velocityY += dino.gravity * dt;
  dino.y += dino.velocityY * dt;

  if (dino.y >= 200) {
    dino.y = 200;
    dino.velocityY = 0;
    dino.grounded = true;
    dino.jumpsRemaining = dino.maxJumps;
  }

  // Spawn Timers
  obstacleTimer += dt;
  if (obstacleTimer > (2.2 / (gameSpeed / 300)) + Math.random() * 0.4) {
    spawnObstacle();
    obstacleTimer = 0;
  }

  orbTimer += dt;
  if (orbTimer > 1.8) {
    spawnXPOrb();
    orbTimer = 0;
  }

  // Obstacle Collision
  for (let i = obstacles.length - 1; i >= 0; i--) {
    let obs = obstacles[i];
    obs.x -= gameSpeed * dt;

    if (
      dino.x < obs.x + obs.width &&
      dino.x + dino.width > obs.x &&
      dino.y < obs.y + obs.height &&
      dino.y + dino.height > obs.y
    ) {
      if (dino.shieldCount > 0) {
        dino.shieldCount--;
        obs.active = false;
      } else {
        triggerGameOver();
      }
    }

    if (obs.x < -50 || !obs.active) obstacles.splice(i, 1);
  }

  // XP Orb Collection & Magnet Logic
  for (let i = xpOrbs.length - 1; i >= 0; i--) {
    let orb = xpOrbs[i];
    orb.x -= gameSpeed * dt;

    // Magnet pulling effect
    if (dino.magnetRadius > 0) {
      const dx = (dino.x + dino.width / 2) - orb.x;
      const dy = (dino.y + dino.height / 2) - orb.y;
      const dist = Math.sqrt(dx * dx + dy * dy);

      if (dist < dino.magnetRadius) {
        orb.x += (dx / dist) * 400 * dt;
        orb.y += (dy / dist) * 400 * dt;
      }
    }

    // Collection AABB
    if (
      dino.x < orb.x + orb.size &&
      dino.x + dino.width > orb.x &&
      dino.y < orb.y + orb.size &&
      dino.y + dino.height > orb.y
    ) {
      addXP(orb.value);
      xpOrbs.splice(i, 1);
    } else if (orb.x < -50) {
      xpOrbs.splice(i, 1);
    }
  }
}

function triggerGameOver() {
  gameOver = true;
  ctx.fillStyle = 'rgba(0,0,0,0.8)';
  ctx.fillRect(0, 0, canvas.width, canvas.height);
  
  ctx.fillStyle = '#ff4757';
  ctx.font = 'bold 30px sans-serif';
  ctx.textAlign = 'center';
  ctx.fillText('RUN TERMINATED', canvas.width / 2, canvas.height / 2 - 20);
  
  ctx.fillStyle = '#fff';
  ctx.font = '16px sans-serif';
  ctx.fillText(`Final Distance: ${Math.floor(distance)}m  |  Level Reached: ${level}`, canvas.width / 2, canvas.height / 2 + 15);
  ctx.fillText('Press SPACE or TAP to restart', canvas.width / 2, canvas.height / 2 + 45);
}

function render() {
  ctx.clearRect(0, 0, canvas.width, canvas.height);

  // Dynamic Biomes based on Level
  const biomeHue = (level * 40) % 360;
  ctx.fillStyle = `hsl(${biomeHue}, 20%, 8%)`;
  ctx.fillRect(0, 0, canvas.width, canvas.height);

  // Ground Line
  ctx.strokeStyle = `hsl(${biomeHue}, 40%, 30%)`;
  ctx.lineWidth = 3;
  ctx.beginPath();
  ctx.moveTo(0, 250);
  ctx.lineTo(canvas.width, 250);
  ctx.stroke();

  // Magnet Aura (if active)
  if (dino.magnetRadius > 0) {
    ctx.strokeStyle = 'rgba(0, 210, 255, 0.15)';
    ctx.beginPath();
    ctx.arc(dino.x + dino.width/2, dino.y + dino.height/2, dino.magnetRadius, 0, Math.PI * 2);
    ctx.stroke();
  }

  // Player Render
  ctx.fillStyle = '#00d2ff';
  ctx.fillRect(dino.x, dino.y, dino.width, dino.height);

  // Shields Render
  if (dino.shieldCount > 0) {
    ctx.strokeStyle = '#ffd700';
    ctx.lineWidth = 2;
    for (let i = 0; i < dino.shieldCount; i++) {
      ctx.strokeRect(dino.x - 3 - (i * 3), dino.y - 3 - (i * 3), dino.width + 6 + (i * 6), dino.height + 6 + (i * 6));
    }
  }

  // Obstacles
  ctx.fillStyle = '#ff4757';
  obstacles.forEach(obs => ctx.fillRect(obs.x, obs.y, obs.width, obs.height));

  // XP Orbs
  ctx.fillStyle = '#00ff88';
  xpOrbs.forEach(orb => {
    ctx.beginPath();
    ctx.arc(orb.x, orb.y, orb.size, 0, Math.PI * 2);
    ctx.fill();
  });

  // UI Headers & XP Bar
  ctx.fillStyle = '#fff';
  ctx.font = '14px monospace';
  ctx.textAlign = 'left';
  ctx.fillText(`DIST: ${Math.floor(distance)}m`, 20, 25);
  ctx.fillText(`LVL: ${level}`, 160, 25);

  // XP Bar Drawing
  const barWidth = 200;
  const barHeight = 10;
  const xpRatio = Math.min(currentXP / targetXP, 1);

  ctx.fillStyle = '#333';
  ctx.fillRect(240, 15, barWidth, barHeight);
  ctx.fillStyle = '#00ff88';
  ctx.fillRect(240, 15, barWidth * xpRatio, barHeight);
  ctx.strokeStyle = '#555';
  ctx.lineWidth = 1;
  ctx.strokeRect(240, 15, barWidth, barHeight);
}

requestAnimationFrame(gameLoop);
</script>
